请将我上传的文件路径对应填入下方脚本中并直接执行，生成：

```python
import cv2
import numpy as np
from PIL import Image, ImageFilter
import scipy.ndimage as ndimage

def run_lossless_replacement(webpage_path, portrait_path, trans_layer_path):
    # 1. 读取原始图像
    im_web = Image.open(webpage_path).convert("RGB")          # (1215, 618)
    im_port = Image.open(portrait_path).convert("RGB")        # (896, 1200) 或 (1792, 2400)
    im_trans = Image.open(trans_layer_path).convert("RGBA")   # (1133, 481)

    # 统一将 3D 网格人像标准化到 (896, 1200) 并做细节锐化
    if im_port.size != (896, 1200):
        im_port = im_port.resize((896, 1200), Image.LANCZOS)
    im_port = im_port.filter(ImageFilter.UnsharpMask(radius=1.1, percent=120, threshold=2))
    arr_p = np.array(im_port).astype(np.float32)
    hp, wp, _ = arr_p.shape

    # 2. 提取 3D 网格人像的高精度 Alpha 遮罩与核心面部区域
    p_bg_strip = ndimage.gaussian_filter1d(np.median(arr_p[:, :45, :], axis=1), sigma=40, axis=0)
    p_diff = np.linalg.norm(arr_p - p_bg_strip[:, None, :], axis=2)
    lum_p = 0.299*arr_p[:,:,0] + 0.587*arr_p[:,:,1] + 0.114*arr_p[:,:,2]
    lum_bg = 0.299*p_bg_strip[:,0] + 0.587*p_bg_strip[:,1] + 0.114*p_bg_strip[:,2]
    signal = np.maximum(p_diff, np.abs(lum_p - lum_bg[:, None]) * 1.3)
    alpha_raw = np.clip((signal - 5.5) / 32.0, 0.0, 1.0)

    yy, xx = np.mgrid[0:hp, 0:wp]
    core_head = (((xx - 480) / 280.0)**2 + ((yy - 560) / 360.0)**2) < 1.0
    core_neck = (((xx - 560) / 220.0)**2 + ((yy - 980) / 280.0)**2) < 1.0
    core_mask = cv2.GaussianBlur(np.maximum(core_head.astype(np.float32), core_neck.astype(np.float32)), (61, 61), 20)

    alpha_p = np.maximum(alpha_raw, core_mask * 0.98)
    fade_left = np.clip((xx - 20) / 120.0, 0.0, 1.0)
    fade_top = np.clip((yy - 8) / 55.0, 0.0, 1.0)
    alpha_p = np.clip(cv2.GaussianBlur(alpha_p, (9, 9), 2.5) * fade_left * fade_top, 0.0, 1.0)

    # 3. 放大网页原图至 2倍超清 (2430 x 1236)，保留 100% 原始文字锐度
    W2, H2 = 2430, 1236
    web_2x = im_web.resize((W2, H2), Image.LANCZOS).filter(ImageFilter.UnsharpMask(radius=0.8, percent=80, threshold=2))
    arr_w2 = np.array(web_2x).astype(np.float32)
    canvas = arr_w2.copy()

    # 4. 仅在 X >= 1680 区域擦除原白发女孩（严禁触碰 X <= 1610 的中央文字！）
    bg_strip_w2 = ndimage.gaussian_filter1d(np.median(arr_w2[:, 1615:1680, :], axis=1), sigma=25, axis=0)
    clean_bg_2x = np.tile(bg_strip_w2[:, None, :], (1, W2, 1))

    erase_mask = np.zeros((H2, W2), dtype=np.float32)
    erase_mask[210:1195, 1695:2398] = 1.0
    erase_mask[145:305, 2135:2385] = 0.0  # 保护右上角 A MORE BEAUTIFUL YOU.
    erase_mask = cv2.GaussianBlur(erase_mask, (51, 51), 16)
    erase_mask[:, :1635] = 0.0            # 绝对保护中央文字（最右端 X=1599）
    erase_mask[:96, :] = 0.0              # 绝对保护顶部导航栏
    erase_mask[1206:, :] = 0.0            # 绝对保护底部卡片
    erase_mask[:, 2401:] = 0.0            # 绝对保护右侧滚动条

    for c in range(3):
        canvas[:, :, c] = canvas[:, :, c] * (1.0 - erase_mask) + clean_bg_2x[:, :, c] * erase_mask

    # 5. 将高清 3D 网格人像融合至右侧 (X: [1613, 2401], Y: [154, 1206])
    tw_p, th_p = 788, 1052
    px0, py0 = 2401 - tw_p, 1206 - th_p   # px0 = 1613 (>1599，完全不压中央文字！), py0 = 154
    port_s = cv2.resize(arr_p, (tw_p, th_p), interpolation=cv2.INTER_LANCZOS4)
    alpha_s = cv2.resize(alpha_p, (tw_p, th_p), interpolation=cv2.INTER_LINEAR)
    core_s = cv2.resize(core_mask, (tw_p, th_p), interpolation=cv2.INTER_LINEAR)[:, :, None]

    # 背景低频色调自适应对齐，做到零接缝融合
    roi_canvas = canvas[py0:py0+th_p, px0:px0+tw_p].copy()
    port_s_bg = ndimage.gaussian_filter1d(np.median(port_s[:, :35, :], axis=1), sigma=35, axis=0)[:, None, :]
    roi_bg = cv2.GaussianBlur(roi_canvas, (91, 91), 30)
    port_harmonized = np.clip(port_s + (roi_bg - port_s_bg) * (1.0 - 0.85 * core_s), 0, 255)

    yy_s, xx_s = np.mgrid[0:th_p, 0:tw_p]
    soft_box = np.clip((xx_s - 15) / 110.0, 0.0, 1.0) * np.clip((yy_s - 10) / 50.0, 0.0, 1.0)
    blend_mask = np.maximum(alpha_s, soft_box * 0.82) * soft_box

    for c in range(3):
        roi_canvas[:, :, c] = roi_canvas[:, :, c] * (1.0 - blend_mask) + port_harmonized[:, :, c] * blend_mask
    canvas[py0:py0+th_p, px0:px0+tw_p] = roi_canvas

    # 6. 恢复右上角 "A MORE BEAUTIFUL YOU." 文字锐度
    text_orig = arr_w2[145:305, 2135:2385]
    text_lum = 0.299*text_orig[:,:,0] + 0.587*text_orig[:,:,1] + 0.114*text_orig[:,:,2]
    text_alpha = np.clip((cv2.GaussianBlur(text_lum, (31, 31), 10) - text_lum - 2.5) / 22.0, 0.0, 1.0)
    for c in range(3):
        canvas[145:305, 2135:2385, c] = (
            canvas[145:305, 2135:2385, c] * (1.0 - text_alpha) + text_orig[:, :, c] * text_alpha
        )

    # 7. 恢复右下角悬浮组件（yd 图标 + 外白圈），并将上方小圆头像替换为 3D 网格人像
    wy0, wy1, wx0, wx1 = 880, 1025, 2330, 2401
    yy_w, xx_w = np.mgrid[wy0:wy1, wx0:wx1]
    dist_top = np.sqrt((xx_w - 2371.5)**2 + (yy_w - 922.0)**2)
    dist_bot = np.sqrt((xx_w - 2367.1)**2 + (yy_w - 982.6)**2)
    widget_mask = np.clip(33.5 - np.minimum(dist_top, dist_bot), 0.0, 1.0)
    for c in range(3):
        canvas[wy0:wy1, wx0:wx1, c] = (
            canvas[wy0:wy1, wx0:wx1, c] * (1.0 - widget_mask) + arr_w2[wy0:wy1, wx0:wx1, c] * widget_mask
        )

    # 替换上方小圆头像内部 (圆心 2371.5, 922.0, 内圆半径 24px)
    av_r = 24
    av_crop = cv2.resize(arr_p[120:720, 140:740], (av_r*2, av_r*2), interpolation=cv2.INTER_AREA)
    ay0, ax0 = int(922 - av_r), int(2371.5 - av_r)
    yy_a, xx_a = np.mgrid[0:av_r*2, 0:av_r*2]
    av_mask = np.clip(av_r - 0.5 - np.sqrt((xx_a - av_r + 0.5)**2 + (yy_a - av_r + 0.5)**2), 0.0, 1.0)
    for c in range(3):
        canvas[ay0:ay0+av_r*2, ax0:ax0+av_r*2, c] = (
            canvas[ay0:ay0+av_r*2, ax0:ax0+av_r*2, c] * (1.0 - av_mask) + av_crop[:, :, c] * av_mask
        )

    # 保存 2 倍超清完整网页图
    out_web_path = "Replaced_Webpage_UltraHD_2430x1236.png"
    Image.fromarray(np.clip(canvas, 0, 255).astype(np.uint8)).save(out_web_path, format="PNG", compress_level=1)

    # 8. 生成 2 倍超清透明底 PNG 图层 (2266 x 962)
    trans_2x = np.array(im_trans.resize((2266, 962), Image.LANCZOS))
    trans_2x[:, 1500:, :] = 0  # 清空右侧原白发女孩
    tp_w, tp_h = 686, 920
    port_t = cv2.resize(arr_p, (tp_w, tp_h), interpolation=cv2.INTER_LANCZOS4)
    alpha_t = cv2.resize(alpha_p, (tp_w, tp_h), interpolation=cv2.INTER_LINEAR)
    trans_2x[42:42+tp_h, 2266-tp_w:2266, :3] = np.clip(port_t, 0, 255).astype(np.uint8)
    trans_2x[42:42+tp_h, 2266-tp_w:2266, 3] = np.clip(alpha_t * 255.0, 0, 255).astype(np.uint8)

    out_trans_path = "Replaced_Transparent_Layer_UltraHD_2266x962.png"
    Image.fromarray(trans_2x).save(out_trans_path, format="PNG", compress_level=1)
    return out_web_path, out_trans_path
